set(SATTYRE_DOXYFILE_IN ${CMAKE_CURRENT_LIST_DIR}/Doxyfile.in)
set(SATTYRE_DOXYFILE ${CMAKE_CURRENT_BINARY_DIR}/Doxyfile.sattyre)
set(SATTYRE_DOCS_OUTPUT_DIR ${CMAKE_BINARY_DIR}/docs)
set(SATTYRE_DOCS_INSTALL_PREFIX ${CMAKE_BINARY_DIR}/docs-install)

configure_file(${SATTYRE_DOXYFILE_IN} ${SATTYRE_DOXYFILE} @ONLY)

add_custom_target(sattyre-docs ALL
  COMMAND ${DOXYGEN_EXECUTABLE} ${SATTYRE_DOXYFILE}
  WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
  COMMENT "Generating Sattyre documentation with Doxygen"
  VERBATIM
)

install(
  DIRECTORY ${SATTYRE_DOCS_OUTPUT_DIR}/
  DESTINATION ${CMAKE_INSTALL_DOCDIR}
  COMPONENT docs
  OPTIONAL
)

add_custom_target(sattyre-docs-install
  COMMAND ${CMAKE_COMMAND} --install ${CMAKE_BINARY_DIR} --component docs --prefix ${SATTYRE_DOCS_INSTALL_PREFIX}
  DEPENDS sattyre-docs
  COMMENT "Installing generated Sattyre docs into ${SATTYRE_DOCS_INSTALL_PREFIX}"
  VERBATIM
)
